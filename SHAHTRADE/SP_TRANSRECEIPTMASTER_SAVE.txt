ALTER PROCEDURE [dbo].[SP_TRANS_RECEIPTMASTER_SAVE]-- sysname, ProcedureName> 
	(
	@registername varchar(100),
	@ACCdate datetime,
	@ACCNAME varchar(100),
	@name varchar(100),
	@CHQAMT float,
	@CHQNO VARCHAR(50),
	@remarks TEXT,
	@BILLREMARKS VARCHAR(MAX),
	@INWORDS TEXT,
	@RECODATE VARCHAR(15),
	
	@cmpid int,
	@locationid int,
	@userid int,
	@yearid int,
	@transfer bit,

	--PAYMENT GRID PARAMETERS	
	@GRIDSRNO varchar(8000),	
	@PAYTYPE VARCHAR(8000),
	@BILLINITIALS varchar(8000),
	@NARR VARCHAR(8000),
	@AMT VARCHAR(8000),
	
	@AMTPAID VARCHAR(8000),
	@EXTRAAMT VARCHAR(8000),
	@RETURN VARCHAR(8000),
	@BALANCE VARCHAR(8000),
	
	--PAYMENTDESC GRID PARAMETERS	
	@DESCGRIDSRNO varchar(8000),	
	@DESCLEDGERNAME VARCHAR(8000),
	@DESCNARR varchar(8000),
	@DESCAMT VARCHAR(8000),
	@DESCPAYGRIDSRNO VARCHAR(8000),
	@DESCPAYBILLINITIALS VARCHAR(8000),
	@BANKNAME VARCHAR(200)
	
	)
AS

--for GRIDSRNO
declare @spGRIDSRNO int, @strGRIDSRNO TINYINT
declare @spPAYTYPE int, @strPAYTYPE varchar(20)
declare @spBILLINITIALS int, @strBILLINITIALS varchar(20)
declare @spNARR int, @strNARR varchar(200)
declare @spAMT int, @strAMT FLOAT


--for AMTPAID
declare @spAMTPAID int, @strAMTPAID FLOAT
declare @spEXTRAAMT int, @strEXTRAAMT FLOAT
declare @spRETURN int, @strRETURN FLOAT
declare @spBALANCE int, @strBALANCE FLOAT


--for DESCGRIDSRNO
declare @spDESCGRIDSRNO int, @strDESCGRIDSRNO TINYINT
declare @sPDESCLEDGERNAME int, @strDESCLEDGERNAME varchar(100)
declare @sPDESCNARR int, @strDESCNARR varchar(200)
declare @sPDESCAMT int, @strDESCAMT FLOAT
declare @spDESCPAYGRIDSRNO int, @strDESCPAYGRIDSRNO INT
declare @spDESCPAYBILLINITIALS int, @strDESCPAYBILLINITIALS varchar(20)




declare @regid as int, @reginitials varchar(20),@regbillinitials varchar(50), @REGABBR VARCHAR(50)

--getting regid
select @regid = isnull(register_id,0),@reginitials=register_initials, @REGABBR = REGISTER_ABBR from Registermaster where register_name = @registerNAME and register_cmpid = @cmpid AND register_locationid=@locationid AND register_yearid=@yearid  and register_type = 'RECEIPT'




declare @ACCid as int
--getting ACCOUNT id
set @ACCid=SHAHTRADE.dbo.GET_ACC_ID(@ACCNAME,@cmpid,@locationid,@yearid )


declare @ledgerid as int, @PARTYBANKID INT
--getting vendor id
set @ledgerid=SHAHTRADE.dbo.GET_ACC_ID(@name,@cmpid,@locationid,@yearid )
SELECT @PARTYBANKID = PARTYBANKMASTER.PARTYBANK_id FROM PARTYBANKMASTER WHERE PARTYBANK_name = @BANKNAME AND PARTYBANK_yearid = @yearid 



declare @RECEIPT_NO int   --for to get max NUMBER from RECEIPTMASTER
SELECT @RECEIPT_NO = isnull(MAX(RECEIPT_no), 0) + 1 FROM  RECEIPTMASTER where RECEIPT_registerid = @regid AND RECEIPT_cmpid =@cmpid AND receipt_locationid=@locationid AND receipt_yearid=@yearid  

set @regbillinitials=cast(@reginitials as varchar(10)) +'-'+cast(@RECEIPT_NO as varchar(10))



	--adding in RECEIPTMASTER
	insert into RECEIPTMASTER
	values
		(
		@RECEIPT_NO,
		@regid,
		@regbillinitials,						  
		@ACCdate,
		@ACCid,
		@ledgerid,
		@PARTYBANKID,
		@CHQAMT ,	
		@CHQNO,
		@remarks,
		@BILLREMARKS,
		@INWORDS,
		'', --RECODATE
		@cmpid,
		@locationid,
		@userid,
		@yearid,
		@transfer,
		getdate()
		)


begin 

	--FOR GRIDPAYMENT
	while @AMT	<> '' 
	begin					
			
		
		--FOR GRIDSRNO
		SET @spGRIDSRNO = CHARINDEX(',', @GRIDSRNO) 
        IF @spGRIDSRNO>0 
            BEGIN 
                SET @strGRIDSRNO = CAST(LEFT(@GRIDSRNO, @spGRIDSRNO-1) AS TINYINT) 
                SET @GRIDSRNO = RIGHT(@GRIDSRNO, LEN(@GRIDSRNO)-@spGRIDSRNO) 
            END 
        ELSE 
            BEGIN 
                SET @strGRIDSRNO = CAST(@GRIDSRNO AS TINYINT) 
                SET @GRIDSRNO = '' 
            END
		
		
		
		--FOR PAYTYPE
		SET @spPAYTYPE = CHARINDEX(',', @PAYTYPE) 
        IF @spPAYTYPE>0 
            BEGIN 
                SET @strPAYTYPE = CAST(LEFT(@PAYTYPE, @spPAYTYPE-1) AS VARCHAR(20)) 
                SET @PAYTYPE = RIGHT(@PAYTYPE, LEN(@PAYTYPE)-@spPAYTYPE) 
            END 
        ELSE 
            BEGIN 
                SET @strPAYTYPE = CAST(@PAYTYPE AS VARCHAR(20)) 
                SET @PAYTYPE = '' 
            END
		
		
		
		--FOR BILLINITIALS
		SET @spBILLINITIALS = CHARINDEX(',', @BILLINITIALS) 
        IF @spBILLINITIALS>0 
            BEGIN 
                SET @strBILLINITIALS = CAST(LEFT(@BILLINITIALS, @spBILLINITIALS-1) AS VARCHAR(20)) 
                SET @BILLINITIALS = RIGHT(@BILLINITIALS, LEN(@BILLINITIALS)-@spBILLINITIALS) 
            END 
        ELSE 
            BEGIN 
                SET @strBILLINITIALS = CAST(@BILLINITIALS AS VARCHAR(20)) 
                SET @BILLINITIALS = '' 
            END
            
        
        
        --FOR NARR
		SET @spNARR = CHARINDEX(',', @NARR) 
        IF @spNARR>0 
            BEGIN 
                SET @strNARR = CAST(LEFT(@NARR, @spNARR-1) AS VARCHAR(200)) 
                SET @NARR = RIGHT(@NARR, LEN(@NARR)-@spNARR) 
            END 
        ELSE 
            BEGIN 
                SET @strNARR = CAST(@NARR AS VARCHAR(200)) 
                SET @NARR = '' 
            END
        
        
            
		--FOR Amt
		SET @spAMT = CHARINDEX(',', @AMT) 
        IF @spAMT>0 
            BEGIN 
                SET @strAMT = CAST(LEFT(@AMT, @spAMT-1) AS FLOAT) 
                SET @AMT = RIGHT(@AMT, LEN(@AMT)-@spAMT) 
            END 
        ELSE 
            BEGIN 
                SET @strAMT = CAST(@AMT AS FLOAT) 
                SET @AMT = '' 
            END 
		

		 
		
	
				
       INSERT INTO  RECEIPTMASTER_DESC
		 values 
		(
		@RECEIPT_NO,
		@regid,
		@regbillinitials,
		CAST(@strGRIDSRNO  as TINYINT),
		CAST(@strPAYTYPE AS VARCHAR(20)),		
		CAST(@strBILLINITIALS AS VARCHAR(20)),
		CAST(@strNARR AS VARCHAR(200) ),						
		CAST(@strAMT AS FLOAT ),							
		0,							--AMTPAID
		0,							--EXTRAAMT
		0,							--RETURN
		CAST(@strAMT AS FLOAT ),	--BALANCE
		@cmpid ,
		@locationid ,
		@userid ,	    
		@yearid  ,	     	    
	    @transfer ,
		getdate()		
		)  
		
		
		
		IF CAST(@strPAYTYPE AS VARCHAR(15)) = 'Against Bill'
		BEGIN
			--FINDING BILLTYPE AND UPDATING BILLS AS PAID
			DECLARE @BTYPE VARCHAR(100)
			SELECT @BTYPE = T.BILLTYPE FROM (SELECT BILL_INITIALS AS BILLINITIALS, BILL_BALANCE AS BALANCE, BILL_CMPID AS CMPID, BILL_YEARID AS YEARID, 'OPENING' AS BILLTYPE FROM OPENINGBILL UNION ALL SELECT INVOICEMASTER.INVOICE_INITIALS AS BILLINITIALS, INVOICEMASTER.INVOICE_BALANCE AS BALANCE, INVOICEMASTER.INVOICE_CMPID AS CMPID, INVOICEMASTER.INVOICE_YEARID AS YEARID, 'INVOICE' AS BILLTYPE FROM INVOICEMASTER) AS T WHERE T.BILLINITIALS = CAST(@strBILLINITIALS AS VARCHAR(50)) AND T.CMPID = @cmpid AND T.YEARID = @yearid 
			IF @BTYPE = 'OPENING'
			BEGIN
				UPDATE OPENINGBILL SET BILL_AMTPAIDREC = BILL_AMTPAIDREC + CAST(@strAMT AS FLOAT ), BILL_BALANCE = BILL_BALANCE - CAST(@strAMT AS FLOAT ) WHERE BILL_LEDGERID=@ledgerid AND BILL_INITIALS = CAST(@strBILLINITIALS AS VARCHAR(50)) AND BILL_CMPID = @cmpid   AND BILL_LOCATIONID=@locationid AND BILL_YEARID = @yearid 
			END
			IF @BTYPE = 'INVOICE'
			BEGIN
				UPDATE INVOICEMASTER SET INVOICE_AMTREC = INVOICE_AMTREC + CAST(@strAMT AS FLOAT ), INVOICE_BALANCE = INVOICE_BALANCE - CAST(@strAMT AS FLOAT ) WHERE INVOICE_INITIALS = CAST(@strBILLINITIALS AS VARCHAR(50)) AND INVOICE_CMPID = @cmpid   AND INVOICE_YEARID = @yearid 
			END
		END
	end
	
	
	
	--FOR GRIDPAYMENTDESC
	while @DESCAMT	<> '' 
	begin					
			
		
		--FOR DESCGRIDSRNO
		SET @spDESCGRIDSRNO = CHARINDEX(',', @DESCGRIDSRNO) 
        IF @spDESCGRIDSRNO>0 
            BEGIN 
                SET @strDESCGRIDSRNO = CAST(LEFT(@DESCGRIDSRNO, @spDESCGRIDSRNO-1) AS TINYINT) 
                SET @DESCGRIDSRNO = RIGHT(@DESCGRIDSRNO, LEN(@DESCGRIDSRNO)-@spDESCGRIDSRNO) 
            END 
        ELSE 
            BEGIN 
                SET @strDESCGRIDSRNO = CAST(@DESCGRIDSRNO AS TINYINT) 
                SET @DESCGRIDSRNO = '' 
            END
		
		
		
		--FOR DESCLEDGERNAME
		SET @spDESCLEDGERNAME = CHARINDEX(',', @DESCLEDGERNAME) 
        IF @spDESCLEDGERNAME>0 
            BEGIN 
                SET @strDESCLEDGERNAME = CAST(LEFT(@DESCLEDGERNAME, @spDESCLEDGERNAME-1) AS VARCHAR(200)) 
                SET @DESCLEDGERNAME = RIGHT(@DESCLEDGERNAME, LEN(@DESCLEDGERNAME)-@spDESCLEDGERNAME) 
            END 
        ELSE 
            BEGIN 
                SET @strDESCLEDGERNAME = CAST(@DESCLEDGERNAME AS VARCHAR(200)) 
                SET @DESCLEDGERNAME = '' 
            END
		
		
		
		--FOR DESCNARR
		SET @spDESCNARR = CHARINDEX(',', @DESCNARR) 
        IF @spDESCNARR>0 
            BEGIN 
                SET @strDESCNARR = CAST(LEFT(@DESCNARR, @spDESCNARR-1) AS VARCHAR(200)) 
                SET @DESCNARR = RIGHT(@DESCNARR, LEN(@DESCNARR)-@spDESCNARR) 
            END 
        ELSE 
            BEGIN 
                SET @strDESCNARR = CAST(@DESCNARR AS VARCHAR(200)) 
                SET @DESCNARR = '' 
            END
            
        
        
        --FOR DESCAMT
		SET @spDESCAMT = CHARINDEX(',', @DESCAMT) 
        IF @spDESCAMT>0 
            BEGIN 
                SET @strDESCAMT = CAST(LEFT(@DESCAMT, @spDESCAMT-1) AS FLOAT) 
                SET @DESCAMT = RIGHT(@DESCAMT, LEN(@DESCAMT)-@spDESCAMT) 
            END 
        ELSE 
            BEGIN 
                SET @strDESCAMT = CAST(@DESCAMT AS FLOAT) 
                SET @DESCAMT = '' 
            END
            
	
		--FOR DESCPAYGRIDSRNO
		SET @spDESCPAYGRIDSRNO = CHARINDEX(',', @DESCPAYGRIDSRNO) 
        IF @spDESCPAYGRIDSRNO>0 
            BEGIN 
                SET @strDESCPAYGRIDSRNO = CAST(LEFT(@DESCPAYGRIDSRNO, @spDESCPAYGRIDSRNO-1) AS TINYINT) 
                SET @DESCPAYGRIDSRNO = RIGHT(@DESCPAYGRIDSRNO, LEN(@DESCPAYGRIDSRNO)-@spDESCPAYGRIDSRNO) 
            END 
        ELSE 
            BEGIN 
                SET @strDESCPAYGRIDSRNO = CAST(@DESCPAYGRIDSRNO AS TINYINT) 
                SET @DESCPAYGRIDSRNO = '' 
            END 
            
            
        --FOR DESCPAYBILLINITIALS
		SET @spDESCPAYBILLINITIALS = CHARINDEX(',', @DESCPAYBILLINITIALS) 
        IF @spDESCPAYBILLINITIALS>0 
            BEGIN 
                SET @strDESCPAYBILLINITIALS = CAST(LEFT(@DESCPAYBILLINITIALS, @spDESCPAYBILLINITIALS-1) AS VARCHAR(20)) 
                SET @DESCPAYBILLINITIALS = RIGHT(@DESCPAYBILLINITIALS, LEN(@DESCPAYBILLINITIALS)-@spDESCPAYBILLINITIALS) 
            END 
        ELSE 
            BEGIN 
                SET @strDESCPAYBILLINITIALS = CAST(@DESCPAYBILLINITIALS AS VARCHAR(20)) 
                SET @DESCPAYBILLINITIALS = '' 
            END 
		
		
		DECLARE @DESCLEDGERID INT
		select @DESCLEDGERID = SHAHTRADE.dbo.GET_ACC_ID(CAST(@strDESCLEDGERNAME AS VARCHAR(200)) ,@cmpid, @locationid ,@yearid )

	
				
       INSERT INTO  RECEIPTMASTER_GRIDDESC
		 values 
		(
		@RECEIPT_NO,
		@regid,
		@regbillinitials,
		CAST(@strDESCGRIDSRNO  as TINYINT),
		@DESCLEDGERID,
		CAST(@strDESCNARR AS VARCHAR(200) ),						
		CAST(@strDESCAMT AS FLOAT ),						
		CAST(@strDESCPAYGRIDSRNO AS TINYINT),						
		CAST(@strDESCPAYBILLINITIALS AS VARCHAR(20)),						
		@cmpid ,
		@locationid ,
		@userid ,	    
		@yearid  ,	     	    
	    @transfer ,
		getdate()		
		)  
		
		
		--FINDING BILLTYPE AND UPDATING BILLS AS PAID
		SELECT @BTYPE = T.BILLTYPE FROM (SELECT BILL_INITIALS AS BILLINITIALS, BILL_BALANCE AS BALANCE, BILL_CMPID AS CMPID, BILL_YEARID AS YEARID, 'OPENING' AS BILLTYPE FROM OPENINGBILL UNION ALL SELECT INVOICEMASTER.INVOICE_INITIALS AS BILLINITIALS, INVOICEMASTER.INVOICE_BALANCE AS BALANCE, INVOICEMASTER.INVOICE_CMPID AS CMPID, INVOICEMASTER.INVOICE_YEARID AS YEARID, 'INVOICE' AS BILLTYPE FROM INVOICEMASTER ) AS T WHERE T.BILLINITIALS = CAST(@strDESCPAYBILLINITIALS AS VARCHAR(100)) AND T.CMPID = @cmpid AND T.YEARID = @yearid 
		IF @BTYPE = 'OPENING'
			BEGIN
				UPDATE OPENINGBILL SET BILL_EXTRAAMT  = BILL_EXTRAAMT + CAST(@strDESCAMT AS FLOAT ), BILL_BALANCE = BILL_BALANCE - CAST(@strDESCAMT AS FLOAT ) WHERE BILL_LEDGERID=@ledgerid AND BILL_INITIALS = CAST(@strDESCPAYBILLINITIALS AS VARCHAR(50)) AND BILL_CMPID = @cmpid   AND BILL_LOCATIONID=@locationid AND BILL_YEARID = @yearid 
			END
		IF @BTYPE = 'INVOICE'
			BEGIN
				UPDATE INVOICEMASTER SET INVOICE_EXTRAAMT = INVOICE_EXTRAAMT + CAST(@strDESCAMT AS FLOAT ), INVOICE_BALANCE = INVOICE_BALANCE - CAST(@strDESCAMT AS FLOAT ) WHERE INVOICE_INITIALS = CAST(@strDESCPAYBILLINITIALS AS VARCHAR(50)) AND INVOICE_CMPID = @cmpid  AND INVOICE_YEARID = @yearid 
			END
		
			
		--ACCOUNTS ENTYRY
		BEGIN
		
			--***************** PARTY NAME(CREDIT) AND DESCLEDGERID(DEBIT) **********************
			INSERT INTO ACCMASTER
			VALUES
			(
				@ledgerid,
				CAST(@strDESCAMT AS FLOAT ),
				@DESCLEDGERID,
				@RECEIPT_NO,
				@ACCdate,
				'RECEIPT',
				@regabbr,
				@regbillinitials,
				@cmpid ,
				@locationid ,
				@userid ,	    
				@yearid  ,	     	    
				@transfer ,
				getdate()	
			)
		
		END
		
		
		
	end
	
end
begin

	--ADDING IN ACCMASTER	
    --***************** PARTY NAME(CREDIT) AND ACCOUNTS(DEBIT) **********************
	INSERT INTO ACCMASTER
	VALUES
	(
		@ledgerid,
		@CHQAMT,
		@ACCid,
		@RECEIPT_NO,
		@ACCdate ,
		'RECEIPT',
		@regabbr,
		@regbillinitials,
		@cmpid ,
		@locationid ,
		@userid ,	    
		@yearid  ,	     	    
	    @transfer ,
		getdate()	
	)
	
	SELECT @RECEIPT_NO 
	
end
