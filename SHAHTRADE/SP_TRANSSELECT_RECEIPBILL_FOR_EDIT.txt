ALTER PROCEDURE [dbo].[SP_TRANS_SELECT_RECEIPTBILL_FOR_EDIT]-- sysname, ProcedureName> 
	(
	@RECNO INT,
	@REGISTERNAME VARCHAR(100),
	@CMPID INT,
	@LOCATIONID INT,
	@YEARID INT
	)
AS


begin 

SELECT     RECEIPTMASTER.receipt_no AS RECNO, REGISTERMASTER.register_name AS REGISTERNAME, RECEIPTMASTER.receipt_date AS ACCDATE, 
                      ACCLEDGER.Acc_cmpname AS ACCNAME, LEDGERS.Acc_cmpname AS LEDGERNAME, RECEIPTMASTER.receipt_total AS CHQAMT, 
                      ISNULL(RECEIPTMASTER.RECEIPT_CHQNO, '') AS CHQNO, ISNULL(RECEIPTMASTER.receipt_remarks, '') AS REMARKS, 
                      RECEIPTMASTER_DESC.receipt_gridsrno AS GRIDSRNO, RECEIPTMASTER_DESC.receipt_paytype AS PAYTYPE, 
                      RECEIPTMASTER_DESC.RECEIPT_BILLINITIALS AS BILLINITIALS, RECEIPTMASTER_DESC.RECEIPT_GRIDREMARKS AS NARR, 
                      RECEIPTMASTER_DESC.receipt_amt AS AMT, ISNULL(RECEIPTMASTER.RECEIPT_RECODATE, '') AS RECODATE, ISNULL(PARTYBANKMASTER.PARTYBANK_name, '') 
                      AS BANKNAME
FROM         RECEIPTMASTER INNER JOIN
                      RECEIPTMASTER_DESC ON RECEIPTMASTER.receipt_no = RECEIPTMASTER_DESC.receipt_no AND 
                      RECEIPTMASTER.receipt_yearid = RECEIPTMASTER_DESC.receipt_yearid AND 
                      RECEIPTMASTER.receipt_registerid = RECEIPTMASTER_DESC.receipt_registerid INNER JOIN
                      LEDGERS ON RECEIPTMASTER.receipt_ledgerid = LEDGERS.Acc_id INNER JOIN
                      REGISTERMASTER ON RECEIPTMASTER.receipt_registerid = REGISTERMASTER.register_id INNER JOIN
                      LEDGERS AS ACCLEDGER ON RECEIPTMASTER.receipt_accid = ACCLEDGER.Acc_id LEFT OUTER JOIN
                      PARTYBANKMASTER ON RECEIPTMASTER.RECEIPT_BANKNAMEID = PARTYBANKMASTER.PARTYBANK_id
WHERE     (RECEIPTMASTER.receipt_no = @RECNO) AND (REGISTERMASTER.register_name = @REGISTERNAME) AND (RECEIPTMASTER.receipt_cmpid = @CMPID) AND 
                      (RECEIPTMASTER.receipt_locationid = @LOCATIONID) AND (RECEIPTMASTER.receipt_yearid = @YEARID)
end
