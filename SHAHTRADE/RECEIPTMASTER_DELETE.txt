ALTER PROCEDURE [dbo].[SP_TRANS_RECEIPTMASTER_DELETE]-- sysname, ProcedureName> 
	(
	@RECNO INT,
	@registername varchar(100),
	@cmpid int,
	@LOCATIONID INT,
	@YEARID INT
	)
AS


DECLARE @BINIT AS VARCHAR(50)
DECLARE @BTYPE VARCHAR(100)

declare @regid as int
declare @regABBR as VARCHAR(100)

--getting regid
select @regid = isnull(register_id,0), @regABBR = register_abbr  from Registermaster where register_name = @registerNAME and register_cmpid = @cmpid AND register_locationid = @LOCATIONID AND register_yearid = @YEARID  and register_type = 'RECEIPT'


DECLARE @LEDGERID INT
SELECT @LEDGERID = ISNULL(RECEIPT_LEDGERID,0) FROM RECEIPTMASTER WHERE receipt_no = @RECNO AND receipt_registerid = @regid AND receipt_cmpid = @cmpid AND receipt_locationid = @LOCATIONID AND receipt_yearid = @YEARID 


BEGIN TRANSACTION

	--deleting from RECEIPT master 
	begin
		delete from RECEIPTMASTER 
		where RECEIPT_no=@RECNO AND RECEIPT_registerid = @REGID and RECEIPT_cmpid=@cmpid AND receipt_locationid = @LOCATIONID AND receipt_yearid  = @YEARID 
	end

	--UNDO THE DATA FROM RECEIPTDESCRIPTION INTO RESPECTIVE TABLES
	--before deleting RECEIPTDESCRIPTION undo AMT in RESPECTIVE TABLES table IN INVOICE_AMTREC
	begin
		DECLARE @RECINI VARCHAR(100)
		declare @RECAMT FLOAT
		declare CUR_MAIN cursor for select RECEIPTMASTER_DESC.RECEIPT_BILLINITIALS,  RECEIPTMASTER_DESC.RECEIPT_AMT from RECEIPTMASTER_DESC where RECEIPTMASTER_DESC.RECEIPT_no = @RECNO AND RECEIPTMASTER_DESC.RECEIPT_REGISTERID = @REGID AND RECEIPTMASTER_DESC.RECEIPT_PAYTYPE = 'Against Bill'  AND RECEIPTMASTER_DESC.RECEIPT_CMPID = @CMPID AND RECEIPTMASTER_DESC.RECEIPT_LOCATIONID = @LOCATIONID AND RECEIPTMASTER_DESC.RECEIPT_YEARID = @YEARID
		open CUR_MAIN
			fetch next from CUR_MAIN into @RECINI, @RECAMT
			while @@Fetch_STATUS = 0
				begin
				
					--FINDING BILLTYPE AND UPDATING BILLS AS PAID
					SELECT @BTYPE = T.BILLTYPE FROM (SELECT BILL_INITIALS AS BILLINITIALS, BILL_BALANCE AS BALANCE, BILL_CMPID AS CMPID, BILL_YEARID AS YEARID, 'OPENING' AS BILLTYPE FROM OPENINGBILL UNION ALL SELECT INVOICEMASTER.INVOICE_INITIALS AS BILLINITIALS, INVOICEMASTER.INVOICE_BALANCE AS BALANCE, INVOICEMASTER.INVOICE_CMPID AS CMPID, INVOICEMASTER.INVOICE_YEARID AS YEARID, 'INVOICE' AS BILLTYPE FROM INVOICEMASTER ) AS T WHERE T.BILLINITIALS = @RECINI AND T.CMPID = @cmpid AND T.YEARID = @yearid 
					IF @BTYPE = 'OPENING'
						BEGIN
							update OPENINGBILL set BILL_AMTPAIDREC =  BILL_AMTPAIDREC - @RECAMT, bill_BALANCE = BILL_BALANCE + @RECAMT
							where BILL_LEDGERID=@ledgerid AND BILL_INITIALS = @RECINI and BILL_CMPID = @CMPID AND BILL_LOCATIONID=@locationid AND BILL_YEARID = @yearid 
						END
					IF @BTYPE = 'INVOICE'
						BEGIN
							update INVOICEMASTER set INVOICE_AMTREC =  INVOICE_AMTREC - @RECAMT, INVOICE_BALANCE = INVOICE_BALANCE + @RECAMT
							where INVOICE_INITIALS = @RECINI and INVOICE_CMPID = @CMPID AND INVOICE_yearid = @yearid 
						END
					
					fetch next from CUR_MAIN into @RECINI, @RECAMT
				end
		close CUR_MAIN
		deallocate CUR_MAIN

	end

						
	--deleting from RECEIPTDESCRIPTION table
	begin
	delete from RECEIPTMASTER_DESC
			where RECEIPT_no=@RECNO AND RECEIPT_registerid = @regid  and RECEIPT_cmpid=@cmpid AND receipt_locationid = @locationid AND receipt_yearid = @yearid 
	end





	--UNDO THE DATA FROM RECEIPTGRIDDESCRIPTION INTO RESPECTIVE TABLES
	--before deleting RECEIPTGRIDDESCRIPTION undo AMT in RESPECTIVE TABLES table IN INVOICE_AMTREC
	begin
		
		declare @RECEXTRAAMT FLOAT
		declare CUR_MAIN cursor for select RECEIPTMASTER_GRIDDESC.RECEIPT_PAYBILLINITIALS,  RECEIPTMASTER_GRIDDESC.RECEIPT_DESCAMT from RECEIPTMASTER_GRIDDESC where RECEIPTMASTER_GRIDDESC.RECEIPT_no = @RECNO AND RECEIPTMASTER_GRIDDESC.RECEIPT_REGISTERID = @REGID AND RECEIPTMASTER_GRIDDESC.RECEIPT_CMPID = @CMPID AND RECEIPTMASTER_GRIDDESC.RECEIPT_LOCATIONID = @LOCATIONID AND RECEIPTMASTER_GRIDDESC.RECEIPT_YEARID = @YEARID
		open CUR_MAIN
			fetch next from CUR_MAIN into @RECINI, @RECEXTRAAMT
			while @@Fetch_STATUS = 0
				begin
				
					--FINDING BILLTYPE AND UPDATING BILLS AS PAID
					SELECT @BTYPE = T.BILLTYPE FROM (SELECT BILL_INITIALS AS BILLINITIALS, BILL_BALANCE AS BALANCE, BILL_CMPID AS CMPID, BILL_YEARID AS YEARID, 'OPENING' AS BILLTYPE FROM OPENINGBILL UNION ALL SELECT INVOICEMASTER.INVOICE_INITIALS AS BILLINITIALS, INVOICEMASTER.INVOICE_BALANCE AS BALANCE, INVOICEMASTER.INVOICE_CMPID AS CMPID, INVOICEMASTER.INVOICE_YEARID AS YEARID, 'INVOICE' AS BILLTYPE FROM INVOICEMASTER) AS T WHERE T.BILLINITIALS = @RECINI AND T.CMPID = @cmpid AND T.YEARID = @yearid 
					IF @BTYPE = 'OPENING'
						BEGIN
							update OPENINGBILL set BILL_EXTRAAMT =  BILL_EXTRAAMT - @RECEXTRAAMT, BILL_BALANCE = BILL_BALANCE + @RECEXTRAAMT
							where BILL_LEDGERID=@ledgerid AND BILL_INITIALS = @RECINI and BILL_CMPID = @CMPID AND BILL_LOCATIONID=@locationid AND BILL_YEARID = @yearid 
						END
					IF @BTYPE = 'INVOICE'
						BEGIN
							update INVOICEMASTER set INVOICE_EXTRAAMT =  INVOICE_EXTRAAMT - @RECEXTRAAMT, INVOICE_BALANCE = INVOICE_BALANCE + @RECEXTRAAMT
							where INVOICE_INITIALS = @RECINI and INVOICE_CMPID = @CMPID AND INVOICE_yearid = @yearid 
						END
					
					fetch next from CUR_MAIN into @RECINI, @RECEXTRAAMT
				end
		close CUR_MAIN
		deallocate CUR_MAIN

	end


						
	--deleting from RECEIPTGRIDDESCRIPTION table
	begin
	delete from RECEIPTMASTER_GRIDDESC
			where RECEIPT_no=@RECNO AND RECEIPT_registerid = @regid  and RECEIPT_cmpid=@cmpid AND receipt_locationid = @locationid AND receipt_yearid = @yearid 
	END



	--deleting from ACCMASTER table
	begin
	delete from ACCMASTER
		where acc_regtype=@REGABBR and acc_billno = @RECNO AND acc_cmpid=@cmpid AND acc_locationid = @LOCATIONID AND acc_yearid = @YEARID 
		SELECT 'Receipt Voucher No. ' + cast(@RECNO as varchar(100)) + ' deleted Successfully' 
	end

	IF @@ERROR != 0 
		BEGIN
			ROLLBACK TRANSACTION
		END

	ELSE
		BEGIN
			COMMIT TRANSACTION
		END
